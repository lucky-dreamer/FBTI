# FBTI (Football TI) 微信小程序 实施计划

> **For agentic workers:** 使用 superpowers:subagent-driven-development 逐任务实现。步骤使用 `- [ ]` 语法追踪进度。

**目标:** 实现一个基于微信云开发的 FBTI 足球人格测试小程序，32 道题匹配 16 位世界杯球星

**架构:** 原生微信小程序 + 微信云开发（云数据库 + 云函数 + 云存储），本地 bundled 题目数据 + 云端记录保存

**技术栈:** WXML + WXSS + JavaScript + 微信云开发 SDK

## 全局约束

- 微信基础库最低版本: 2.20.0
- 云开发环境: 首次部署时用户手动填入 cloudEnv
- 所有用户可见文案使用中文
- 页面不超过 3 个（index, quiz, result）
- 32 道题数据 bundled 在 utils/questions.js
- 16 位球星数据 bundled 在 utils/stars.js
- 包体大小控制在 1MB 以内
- 使用 CSS Animation 实现动画效果，不引入第三方动画库

```javascript
// utils/helpers.js

/**
 * 打乱数组顺序（Fisher-Yates）
 */
function shuffle(arr) {
  const result = [...arr]
  for (let i = result.length - 1; i > 0; i--) {
    const j = Math.floor(Math.random() * (i + 1));
    [result[i], result[j]] = [result[j], result[i]]
  }
  return result
}

/**
 * 进度阶段转换
 * 1-16 -> '上半场', 17-32 -> '下半场', 最后5题 -> '伤停补时'
 */
function getMatchPeriod(current, total) {
  if (current >= total - 4) return '伤停补时'
  if (current <= total / 2) return '上半场'
  return '下半场'
}

/**
 * 格式化匹配度（保留整数）
 */
function formatMatchRate(rate) {
  return Math.min(100, Math.max(0, Math.round(rate * 100)))
}

/**
 * 延迟函数（ms）
 */
function delay(ms) {
  return new Promise(resolve => setTimeout(resolve, ms))
}

module.exports = {
  shuffle,
  getMatchPeriod,
  formatMatchRate,
  delay
}
```

- [ ] **Step 3: 创建 utils/mbti.js（MBTI 计算逻辑）**

```javascript
// utils/mbti.js
const stars = require('./stars')

/**
 * 根据 32 题答案计算 MBTI 类型
 * @param {string[]} answers - 32 个元素的数组，每个元素是 'A' 或 'B'
 * @returns {{ type: string, starId: string, dimensions: object, matchRate: number }}
 */
function calculateType(answers) {
  const questions = require('./questions')
  const scores = { E: 0, I: 0, S: 0, N: 0, T: 0, F: 0, J: 0, P: 0 }

  answers.forEach((answer, i) => {
    const q = questions[i]
    if (answer === 'A') {
      scores[q.dimensionA] += 1
    } else {
      scores[q.dimensionB] += 1
    }
  })

  const dimensions = {
    EI: scores.E >= scores.I ? 'E' : 'I',
    SN: scores.S >= scores.N ? 'S' : 'N',
    TF: scores.T >= scores.F ? 'T' : 'F',
    JP: scores.J >= scores.P ? 'J' : 'P'
  }

  const type = dimensions.EI + dimensions.SN + dimensions.TF + dimensions.JP

  // 匹配度计算：每个维度胜出比例的平均值
  const dimRates = [
    Math.max(scores.E, scores.I) / 8,
    Math.max(scores.S, scores.N) / 8,
    Math.max(scores.T, scores.F) / 8,
    Math.max(scores.J, scores.P) / 8
  ]
  const matchRate = dimRates.reduce((a, b) => a + b, 0) / 4

  // 找到对应球星
  const matched = stars.find(s => s.mbti === type)

  return {
    type,
    starId: matched ? matched.id : 'default',
    dimensions,
    scores,
    matchRate
  }
}

module.exports = { calculateType }
```

```javascript
// cloudfunctions/calculateResult/index.js
const cloud = require('wx-server-sdk')
cloud.init({ env: cloud.DYNAMIC_CURRENT_ENV })
const db = cloud.database()

/**
 * 计算 MBTI 类型并匹配球星
 * 输入: { answers: ['A','B',...] } (32个元素的数组)
 * 输出: { type, star, matchRate, dimensions }
 */
exports.main = async (event, context) => {
  const { answers } = event
  const wxContext = cloud.getWXContext()

  if (!answers || answers.length !== 32) {
    return { code: -1, msg: '答案数量不正确' }
  }

  // 获取所有题目（从云数据库或本地定义）
  const questions = (await db.collection('questions').get()).data

  // 计分
  const scores = { E: 0, I: 0, S: 0, N: 0, T: 0, F: 0, J: 0, P: 0 }
  answers.forEach((answer, i) => {
    const q = questions[i]
    if (!q) return
    if (answer === 'A') {
      scores[q.dimensionA] += 1
    } else {
      scores[q.dimensionB] += 1
    }
  })

  // 算类型
  const type = [
    scores.E >= scores.I ? 'E' : 'I',
    scores.S >= scores.N ? 'S' : 'N',
    scores.T >= scores.F ? 'T' : 'F',
    scores.J >= scores.P ? 'J' : 'P'
  ].join('')

  // 匹配度
  const dimRates = [
    Math.max(scores.E, scores.I) / 8,
    Math.max(scores.S, scores.N) / 8,
    Math.max(scores.T, scores.F) / 8,
    Math.max(scores.J, scores.P) / 8
  ]
  const matchRate = Math.round(dimRates.reduce((a, b) => a + b, 0) / 4 * 100)

  // 查球星
  const stars = (await db.collection('stars').get()).data
  const star = stars.find(s => s.mbti === type) || null

  // 保存记录
  await db.collection('records').add({
    data: {
      openid: wxContext.OPENID,
      result: type,
      matchedStar: star ? star._id : null,
      answers,
      scores,
      matchRate,
      createTime: db.serverDate()
    }
  })

  return {
    code: 0,
    data: {
      type,
      star,
      matchRate,
      dimensions: {
        EI: type[0], SN: type[1], TF: type[2], JP: type[3]
      },
      scores
    }
  }
}
```

```javascript
// cloudfunctions/getStars/index.js
const cloud = require('wx-server-sdk')
cloud.init({ env: cloud.DYNAMIC_CURRENT_ENV })
const db = cloud.database()

exports.main = async (event, context) => {
  const { type } = event
  if (type) {
    // 按 MBTI 类型查询单个球星
    const res = await db.collection('stars').where({ mbti: type }).get()
    return { code: 0, data: res.data[0] || null }
  }
  // 返回所有球星
  const res = await db.collection('stars').get()
  return { code: 0, data: res.data }
}
```

```javascript
// pages/quiz/quiz.js
const questions = require('../../utils/questions')
const { calculateType } = require('../../utils/mbti')
const { getMatchPeriod } = require('../../utils/helpers')

Page({
  data: {
    current: 1,
    total: 32,
    answers: [],
    currentQuestion: null,
    selected: '',
    showNextBtn: false,
    showHalfTime: false,
    showFunFact: false,
    animClass: 'slide-in',
    period: '上半场',
    progressPercent: 0,
    dimLabel: '',
    isStoppageTime: false
  },

  onLoad() {
    // 随机打乱题目顺序，但保持维度分布
    this.questions = this.shuffleQuestions([...questions])
    this.setQuestion(0)
  },

  // 保持维度均衡的洗牌（每个维度内部洗牌，保持8题/维度）
  shuffleQuestions(qs) {
    const dims = { EI: [], SN: [], TF: [], JP: [] }
    qs.forEach(q => dims[q.dimension].push(q))
    Object.keys(dims).forEach(k => {
      for (let i = dims[k].length - 1; i > 0; i--) {
        const j = Math.floor(Math.random() * (i + 1));
        [dims[k][i], dims[k][j]] = [dims[k][j], dims[k][i]]
      }
    })
    return [...dims.EI, ...dims.SN, ...dims.TF, ...dims.JP]
  },

  setQuestion(index) {
    const q = this.questions[index]
    const current = index + 1
    const period = getMatchPeriod(current, 32)
    const progressPercent = (current / 32) * 100
    const dimLabels = { EI: '外倾 vs 内倾', SN: '实感 vs 直觉', TF: '思考 vs 情感', JP: '判断 vs 感知' }

    this.setData({
      current,
      currentQuestion: q,
      selected: '',
      showNextBtn: false,
      showFunFact: false,
      animClass: 'slide-in',
      period,
      progressPercent,
      dimLabel: dimLabels[q.dimension] || '',
      isStoppageTime: current >= 28
    })
  },

  selectOption(e) {
    if (this.data.selected) return
    const opt = e.currentTarget.dataset.opt
    const answers = [...this.data.answers, opt]
    this.setData({
      selected: opt,
      answers: answers
    })

    // 显示彩蛋
    setTimeout(() => {
      this.setData({ showFunFact: true })
    }, 300)

    // 延迟显示下一题按钮
    setTimeout(() => {
      this.setData({ showNextBtn: true })
    }, 600)
  },

  nextQuestion() {
    if (this.data.current === 16) {
      // 第16题后中场休息
      this.setData({ showHalfTime: true })
      return
    }

    if (this.data.current >= 32) {
      this.finishQuiz()
      return
    }

    // 切换到下一题（带滑出动画）
    this.setData({ animClass: 'slide-out' })
    setTimeout(() => {
      this.setQuestion(this.data.current)
    }, 300)
  },

  continueQuiz() {
    this.setData({ showHalfTime: false })
    this.setQuestion(16) // 第17题 (index 16)
  },

  finishQuiz() {
    const result = calculateType(this.data.answers)

    wx.showLoading({ title: '匹配球星中...' })

    // 调用云函数保存记录
    wx.cloud.callFunction({
      name: 'calculateResult',
      data: { answers: this.data.answers }
    }).then(res => {
      wx.hideLoading()
      // 跳转到结果页
      wx.redirectTo({
        url: `/pages/result/result?type=${result.type}&starId=${result.starId}&matchRate=${result.matchRate}`
      })
    }).catch(err => {
      wx.hideLoading()
      // 云函数失败，使用本地计算结果
      wx.redirectTo({
        url: `/pages/result/result?type=${result.type}&starId=${result.starId}&matchRate=${result.matchRate}`
      })
    })
  }
})
```

```javascript
// pages/result/result.js
const stars = require('../../utils/stars')

// 球星emoji映射
const starEmojis = {
  haaland: '👻', lautaro: '🐂', pedri: '🎯', salah: '🐱',
  vandijk: '🛡️', messi: '🐐', yamal: '🌈', debruyne: '🧠',
  mbappe: '⚡', vinicius: '💃', neymar: '🎭', musiala: '🌀',
  kane: '🦁', griezmann: '🎪', bellingham: '👑', ronaldo: '🐐'
}

Page({
  data: {
    star: null,
    matchRate: 0,
    showResult: false,
    starEmoji: '⚽',
    dimensions: []
  },

  onLoad(options) {
    const { type, starId, matchRate } = options

    // 从本地数据找球星
    const star = stars.find(s => s.id === starId)

    if (!star) {
      // 降级：按 MBTI 类型查找
      const star = stars.find(s => s.mbti === type) || stars[0]
    }

    // 构建维度数据
    const dimensions = [
      { label: type[0] === 'E' ? '外倾' : '内倾', value: type[0], percent: 75 },
      { label: type[1] === 'S' ? '实感' : '直觉', value: type[1], percent: 70 },
      { label: type[2] === 'T' ? '思考' : '情感', value: type[2], percent: 80 },
      { label: type[3] === 'J' ? '判断' : '感知', value: type[3], percent: 65 }
    ]

    this.setData({
      star,
      matchRate: parseInt(matchRate) || 85,
      starEmoji: starEmojis[star.id] || '⚽',
      dimensions,
      showResult: true
    })
  },

  retryTest() {
    wx.redirectTo({ url: '/pages/quiz/quiz' })
  },

  savePoster() {
    // 使用 Canvas API 生成海报并保存
    wx.showToast({ title: '海报功能即将上线', icon: 'none' })
  },

  onShareAppMessage() {
    const star = this.data.star
    return {
      title: `我是${star.name}型人格！来测测你的FBTI足球人格 🔥`,
      path: '/pages/index/index',
      imageUrl: '' // 可配置分享封面
    }
  }
})
```
